# Importar librerías de análisis y graficación
import pandas as pd
import matplotlib.pyplot as plt

# 1. Convertir las columnas a tipos numéricos forzando datos inválidos a NaN
df['popularity'] = pd.to_numeric(df['popularity'], errors='coerce')
df['duration_ms'] = pd.to_numeric(df['duration_ms'], errors='coerce')

# 2. Convertir milisegundos a minutos para facilitar la interpretación del eje
df['duration_min'] = df['duration_ms'] / 60000

# 3. Eliminar filas con valores nulos en cualquiera de las dos variables
df_clean = df.dropna(subset=['duration_min', 'popularity']).copy()

# 4. Imprimir la cantidad de registros limpios listos para graficar
print(f"Total de canciones procesadas: {len(df_clean)}")

# 5. Configurar el tamaño del lienzo de la figura
plt.figure(figsize=(10, 6))

# 6. Generar el diagrama de dispersión
plt.scatter(
    df_clean['duration_min'],    # Eje X: Duración de la pista en minutos
    df_clean['popularity'],      # Eje Y: Popularidad (escala 0 a 100)
    alpha=0.4,                   # Nivel de opacidad para apreciar acumulación de datos
    color='purple',              # Color morado idéntico al ejercicio de referencia
    edgecolors='black',          # Contorno negro en cada marcador
    linewidths=0.3,              # Grosor fino para la línea perimetral
    label='Canciones'            # Etiqueta para asociar con plt.legend() sin advertencias
)

# 7. Asignar títulos y etiquetas descriptivas
plt.title('Relación entre Duración (minutos) y Popularidad', fontsize=14)
plt.xlabel('Duración (minutos)', fontsize=12)
plt.ylabel('Popularidad (0 - 100)', fontsize=12)

# 8. Delimitar el eje X entre 0 y 10 minutos para evitar que canciones atípicas deformen la gráfica
plt.xlim(0, 10)

# 9. Trazar la cuadrícula de fondo
plt.grid(True, linestyle='--', alpha=0.4)

# 10. Desplegar la leyenda en la esquina superior derecha
plt.legend(loc='upper right')

# 11. Ajustar márgenes y renderizar el gráfico
plt.tight_layout()
plt.show()

# Importar librerías necesarias
import streamlit as st
import pandas as pd
import matplotlib.pyplot as plt

# 1. Configuración de página
st.set_page_config(page_title="Dashboard Spotify", layout="wide")

# 2. Título de la aplicación
st.title("🎵 Dashboard de Análisis: Spotify Tracks")
st.write("Exploración estadística interactiva del dataset de canciones.")

# 3. Función de carga de datos
@st.cache_data
def cargar_datos():
    data = pd.read_csv('spotify_tracks.csv', encoding='latin-1')
    data.columns = data.columns.str.strip()
    data['popularity'] = pd.to_numeric(data['popularity'], errors='coerce')
    data['duration_ms'] = pd.to_numeric(data['duration_ms'], errors='coerce')
    data['duration_min'] = data['duration_ms'] / 60000
    return data.dropna(subset=['duration_min', 'popularity']).copy()

df = cargar_datos()

# 4. Mostrar el DataFrame
st.header("📋 Vista del DataFrame")
st.dataframe(df.head(100), use_container_width=True)

st.divider()

# 5. Sección de 5 Gráficas
st.header("📊 5 Gráficas de Análisis")

col1, col2 = st.columns(2)

# Gráfica 1: Dispersión
with col1:
    st.subheader("1. Duración vs Popularidad")
    fig1, ax1 = plt.subplots(figsize=(7, 4.5))
    ax1.scatter(df['duration_min'], df['popularity'], alpha=0.35, color='purple', edgecolors='black', linewidths=0.2)
    ax1.set_xlim(0, 10)
    ax1.set_xlabel("Duración (minutos)")
    ax1.set_ylabel("Popularidad (0-100)")
    ax1.grid(True, linestyle='--', alpha=0.3)
    st.pyplot(fig1)

# Gráfica 2: Histograma de Popularidad
with col2:
    st.subheader("2. Distribución de Popularidad")
    fig2, ax2 = plt.subplots(figsize=(7, 4.5))
    ax2.hist(df['popularity'], bins=20, color='#1DB954', edgecolor='black', alpha=0.7)
    ax2.set_xlabel("Popularidad")
    ax2.set_ylabel("Frecuencia")
    ax2.grid(True, linestyle='--', alpha=0.3)
    st.pyplot(fig2)

col3, col4 = st.columns(2)

# Gráfica 3: Top Géneros
with col3:
    st.subheader("3. Top 10 Géneros con más Canciones")
    top_generos = df['genre'].value_counts().head(10)
    fig3, ax3 = plt.subplots(figsize=(7, 4.5))
    top_generos.plot(kind='barh', ax=ax3, color='#4A90E2', edgecolor='black')
    ax3.set_xlabel("Total de Canciones")
    ax3.invert_yaxis()
    ax3.grid(True, linestyle='--', alpha=0.3)
    st.pyplot(fig3)

# Gráfica 4: Contenido Explícito
with col4:
    st.subheader("4. Popularidad Promedio por Contenido Explícito")
    promedio_expl = df.groupby('explicit')['popularity'].mean()
    fig4, ax4 = plt.subplots(figsize=(7, 4.5))
    promedio_expl.plot(kind='bar', ax=ax4, color=['#FFA726', '#E53935'], edgecolor='black')
    ax4.set_xticklabels(['No Explícito', 'Explícito'], rotation=0)
    ax4.set_ylabel("Popularidad Promedio")
    ax4.grid(True, linestyle='--', alpha=0.3)
    st.pyplot(fig4)

# Gráfica 5: Top Artistas
st.subheader("5. Top 10 Artistas con Mayor Presencia")
top_artistas = df['artists'].value_counts().head(10)
fig5, ax5 = plt.subplots(figsize=(11, 4))
top_artistas.plot(kind='bar', ax=ax5, color='#8E24AA', edgecolor='black')
ax5.set_xlabel("Artista")
ax5.set_ylabel("Número de Canciones")
plt.xticks(rotation=45, ha='right')
ax5.grid(True, linestyle='--', alpha=0.3)
st.pyplot(fig5)
